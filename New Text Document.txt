  <script>
  (async function () {
    const { API, esc, toast } = JobFinder;
    if (!await JobFinder.requireRole('seeker')) return;
    const form = document.getElementById('prof-form');
    let profile = null;
    try { profile = (await API.get('/api/profile')).profile; } catch (e) { return; }

    const eduRow = (e = {}) => `<div class="form-row-3 edu-row" style="grid-template-columns:2fr 1fr 1fr 1fr 1fr auto;">
      <input placeholder="Institution" class="e-inst" value="${esc(e.institution || '')}">
      <input placeholder="Degree" class="e-deg" value="${esc(e.degree || '')}">
      <input placeholder="Field" class="e-field" value="${esc(e.field || '')}">
      <input placeholder="From" class="e-start" value="${esc(e.start_year || '')}">
      <input placeholder="To" class="e-end" value="${esc(e.end_year || '')}">
      <button type="button" class="btn btn-ghost btn-sm row-del"><i class="fa-solid fa-xmark"></i></button></div>`;
    const expRow = (x = {}) => `<div class="form-row-3 exp-row" style="grid-template-columns:2fr 2fr 2fr 1fr 1fr auto;">
      <input placeholder="Company" class="x-comp" value="${esc(x.company || '')}">
      <input placeholder="Job title" class="x-title" value="${esc(x.title || '')}">
      <input placeholder="Description" class="x-desc" value="${esc(x.description || '')}">
      <input placeholder="Start (YYYY-MM)" class="x-start" value="${esc(x.start_date || '')}">
      <input placeholder="End" class="x-end" value="${esc(x.end_date || '')}">
      <button type="button" class="btn btn-ghost btn-sm row-del"><i class="fa-solid fa-xmark"></i></button></div>`;

    for (const k of ['full_name','phone','city','country','headline','summary','expected_salary','years_experience','preferred_location','preferred_job_type','linkedin_url'])
      if (form.elements[k] && profile[k] != null) form.elements[k].value = profile[k];
    form.elements.is_public.checked = !!profile.is_public;
    form.elements.skills.value = (profile.skills || []).map(s => s.name).join(', ');
    document.getElementById('edu-list').innerHTML = (profile.education || []).map(eduRow).join('') || eduRow();
    document.getElementById('exp-list').innerHTML = (profile.experience || []).map(expRow).join('') || expRow();

    const list = document.getElementById('resume-list');
    function renderResumes() {
      list.innerHTML = (profile.resumes || []).map(r => `<div class="flex" style="justify-content:space-between;">
        <span><i class="fa-solid fa-file-pdf" style="color:var(--danger);"></i> ${esc(r.original_name)}
        ${r.is_current ? '<span class="badge badge-green">current</span>' : ''}</span>
        <span class="flex"><a class="btn btn-ghost btn-sm" href="/api/resumes/${r.id}/download" target="_blank"><i class="fa-solid fa-eye"></i></a>
        <button class="btn btn-danger btn-sm" data-del-resume="${r.id}"><i class="fa-solid fa-trash"></i></button></span></div>`).join('')
        || '<p class="text-muted">No resume uploaded yet.</p>';
    }
    renderResumes();

    document.getElementById('add-edu').onclick = () => document.getElementById('edu-list').insertAdjacentHTML('beforeend', eduRow());
    document.getElementById('add-exp').onclick = () => document.getElementById('exp-list').insertAdjacentHTML('beforeend', expRow());
    document.body.addEventListener('click', (e) => { if (e.target.closest('.row-del')) e.target.closest('.form-row-3').remove(); });

    form.addEventListener('submit', async (ev) => {
      ev.preventDefault();
      const msg = document.getElementById('prof-msg');
      const body = Object.fromEntries(new FormData(form).entries());
      body.is_public = form.elements.is_public.checked;
      body.education = [...document.querySelectorAll('.edu-row')].map(r => ({
        institution: r.querySelector('.e-inst').value, degree: r.querySelector('.e-deg').value,
        field: r.querySelector('.e-field').value, start_year: r.querySelector('.e-start').value, end_year: r.querySelector('.e-end').value }));
      body.experience = [...document.querySelectorAll('.exp-row')].map(r => ({
        company: r.querySelector('.x-comp').value, title: r.querySelector('.x-title').value,
        description: r.querySelector('.x-desc').value, start_date: r.querySelector('.x-start').value, end_date: r.querySelector('.x-end').value }));
      body.skills = body.skills.split(',').map(s => s.trim()).filter(Boolean);
      try {
        await API.put('/api/profile', body);
        msg.className = 'msg msg-success show'; msg.textContent = 'Profile saved successfully.';
      } catch (err) { msg.className = 'msg msg-error show'; msg.textContent = err.message; }
    });

    document.getElementById('upload-resume').addEventListener('click', async () => {
      const f = document.getElementById('resume-file').files[0];
      if (!f) return toast('Choose a PDF file first.', 'error');
      const fd = new FormData(); fd.append('resume', f);
      try {
        await API.upload('/api/resumes', fd);
        profile = (await API.get('/api/profile')).profile; renderResumes();
        toast('Resume uploaded.', 'success');
      } catch (err) { toast(err.message, 'error'); }
    });
    list.addEventListener('click', async (e) => {
      const id = e.target.closest('[data-del-resume]')?.dataset.delResume; if (!id) return;
      await API.del('/api/resumes/' + id);
      profile.resumes = profile.resumes.filter(r => r.id != id); renderResumes();
    });
    document.getElementById('change-pw').addEventListener('click', async () => {
      const msg = document.getElementById('pw-msg');
      try {
        await API.put('/api/auth/password', { current_password: document.getElementById('pw-cur').value, new_password: document.getElementById('pw-new').value });
        msg.className = 'msg msg-success show'; msg.textContent = 'Password updated.';
      } catch (err) { msg.className = 'msg msg-error show'; msg.textContent = err.message; }
    });
    document.getElementById('delete-account').addEventListener('click', async () => {
      if (!confirm('Really delete your account? This cannot be undone.')) return;
      await API.del('/api/auth/account');
      localStorage.removeItem('jf_csrf'); location.href = '/';
    });
  })();
  </script>

  (async function () {
    const { API, esc, statusBadge } = JobFinder;

    let jobs, received, company;

      API.get('/api/jobs/mine'), API.get('/api/applications/received'), API.get('/api/profile')]); }

        <td><a href="/job-details.html?id=${j.id}">${esc(j.title)}</a></td>
        <td>${statusBadge(j.status)}</td><td>${j.views}</td>

          <a class="btn btn-ghost btn-sm" href="/post-job.html?id=${j.id}"><i class="fa-solid fa-pen"></i></a>
          ${j.status !== 'closed' ? `<button class="btn btn-outline btn-sm" data-status="${j.id}:closed">Close</button>` : ''}
